2. The Combat Simulator Engine (Deterministic Array Processing)

This component handles the 7x5 array allocations, coordinate swaps, match sweeps, and automatic gravity refilling.
python
class SovereignCombatEngine:
    def __init__(self, rows=7, cols=5, seed=42):
        self.rows = rows
        self.cols = cols
        self.seed = seed
        random.seed(self.seed)
        self.board = [[0 for _ in range(cols)] for _ in range(rows)]
        self.generate_clean_board()

    def generate_clean_board(self):
        """Generates a valid initial 7x5 matrix with zero starting matches."""
        for r in range(self.rows):
            for c in range(self.cols):
                while True:
                    element_id = random.choice(list(ELEMENTS.keys()))
                    # Lookback check for horizontal and vertical matches
                    if c >= 2 and self.board[r][c-1] == element_id and self.board[r][c-2] == element_id:
                        continue
                    if r >= 2 and self.board[r-1][c] == element_id and self.board[r-2][c] == element_id:
                        continue
                    self.board[r][c] = element_id
                    break

    def get_board_state(self):
        return [row[:] for row in self.board]

    def execute_swap(self, r1, c1, r2, c2):
        """Validates cell adjacency and performs a deterministic array swap."""
        if abs(r1 - r2) + abs(c1 - c2) != 1:
            return {"status": "INVALID_MOVE", "reason": "Non-adjacent coordinates"}
        
        # Execute tentative memory shift
        self.board[r1][c1], self.board[r2][c2] = self.board[r2][c2], self.board[r1][c1]
        
        matches = self.scan_for_matches()
        if not matches:
            # Revert instantly if no match condition is reached
            self.board[r1][c1], self.board[r2][c2] = self.board[r2][c2], self.board[r1][c1]
            return {"status": "INVALID_MOVE", "reason": "No match created"}
            
        return {"status": "SUCCESS", "initial_matches": matches}

    def scan_for_matches(self):
        """Performs iterative sweeps across the board to detect 3+ configurations."""
        matched_coords = set()
        
        # Horizontal Sweep
        for r in range(self.rows):
            for c in range(self.cols - 2):
                if self.board[r][c] == self.board[r][c+1] == self.board[r][c+2] != 0:
                    matched_coords.update([(r, c), (r, c+1), (r, c+2)])
                    
        # Vertical Sweep
        for c in range(self.cols):
            for r in range(self.rows - 2):
                if self.board[r][c] == self.board[r+1][c] == self.board[r+2][c] != 0:
                    matched_coords.update([(r, c), (r+1, c), (r+2, c)])
                    
        return sorted(list(matched_coords))

    def process_cascade_step(self, matched_coords):
        """Executes lane-clearing tracking, element identification, and gravity shifts."""
        lane_troop_counts = {c: 0 for c in range(self.cols)}
        tile_manifest = []

        # Identify elements before zeroing them out
        for (r, c) in matched_coords:
            element_type = ELEMENTS[self.board[r][c]]
            lane_troop_counts[c] += 1
            tile_manifest.append({"lane": c, "element": element_type})
            self.board[r][c] = 0  # Absolute clearing step

        # Apply downward gravity shifts
        for c in range(self.cols):
            empty_slots = 0
            for r in range(self.rows - 1, -1, -1):
                if self.board[r][c] == 0:
                    empty_slots += 1
                elif empty_slots > 0:
                    self.board[r + empty_slots][c] = self.board[r][c]
                    self.board[r][c] = 0

            # Spawn replacements at the top layer indices
            for r in range(empty_slots):
                self.board[r][c] = random.choice(list(ELEMENTS.keys()))

        return tile_manifest, lane_troop_counts

3. The RPG Metagame and Match Processing Engine

This handles team configurations, target lanes, mana calculations, and the Ghosting Protocol where empty columns double mana speed.
python
class SovereignRosterManager:
    def __init__(self):
        self.heroes = {}  # Assigned by column index 0-4

    def configure_team(self, slot, hero_id, element, mana_speed, attack_power):
        self.heroes[slot] = {
            "hero_id": hero_id,
            "element": element,
            "mana_speed": mana_speed,
            "attack_power": attack_power,
            "current_mana": 0.0,
            "mana_threshold": MANA_SPEED_CHARGES[mana_speed]
        }

class TacticalMatchOrchestrator:
    def __init__(self, combat_engine, roster_manager):
        self.engine = combat_engine
        self.roster = roster_manager
        self.enemies = {
            0: {"id": "BOSS_L", "element": "FIRE", "health": 1500, "active": True},
            2: {"id": "BOSS_C", "element": "ICE", "health": 2500, "active": True},
            4: {"id": "BOSS_R", "element": "NATURE", "health": 1500, "active": True}
        }

    def calculate_lane_target(self, col):
        """Resolves target paths. If a direct column enemy is dead, triggers Ghosting."""
        if col in self.enemies and self.enemies[col]["active"]:
            return col, False # Normal hit
        # Seek closest living enemy
        living_lanes = [lane for lane, data in self.enemies.items() if data["active"]]
        if not living_lanes:
            return None, True
        closest_lane = min(living_lanes, key=lambda x: abs(x - col))
        return closest_lane, True # Ghosting occurs because the direct lane is open

    def execute_turn(self, r1, c1, r2, c2):
        swap_result = self.engine.execute_swap(r1, c1, r2, c2)
        if swap_result["status"] == "INVALID_MOVE":
            return swap_result

        turn_log = []
        cascade_round = 1
        active_matches = swap_result["initial_matches"]

        while active_matches:
            tiles_cleared, lane_counts = self.engine.process_cascade_step(active_matches)
            damage_summary = []
            mana_summary = []

            # Process individual tile effects
            for tile in tiles_cleared:
                lane = tile["lane"]
                element = tile["element"]
                
                target_lane, is_ghosted = self.calculate_lane_target(lane)
                
                # Mana Calculation Engine
                hero_assigned = self.roster.heroes.get(lane)
                if hero_assigned and hero_assigned["element"] == element:
                    # Ghosted tiles multiply processing rate by 2.0x
                    mana_gain = 2.0 if is_ghosted else 1.0
                    hero_assigned["current_mana"] = min(
                        hero_assigned["mana_threshold"], 
                        hero_assigned["current_mana"] + mana_gain
                    )
                    mana_summary.append(f"Hero {hero_assigned['hero_id']} gained {mana_gain} mana.")

                # Combat Damage Processing (Attribute Translation)
                if target_lane is not None:
                    enemy = self.enemies[target_lane]
                    hero_atk = hero_assigned["attack_power"] if hero_assigned else 50
                    mult = ELEMENTAL_MULTIPLIERS[element].get(enemy["element"], 1.0)
                    
                    raw_damage = math.ceil((hero_atk / 5) * mult)
                    enemy["health"] = max(0, enemy["health"] - raw_damage)
                    if enemy["health"] == 0:
                        enemy["active"] = False
                        
                    damage_summary.append(f"Enemy {enemy['id']} hit for {raw_damage} damage ({element}->{enemy['element']}).")

            turn_log.append({
                "cascade_round": cascade_round,
                "tiles_matched_count": len(active_matches),
                "damage_events": damage_summary,
                "mana_events": mana_summary
            })

            # Re-verify layout for nested chains
            active_matches = self.engine.scan_for_matches()
            cascade_round += 1

        return {"status": "TURN_COMPLETE", "cascade_history": turn_log, "final_board": self.engine.get_board_state()}
