    public let clubheadSpeedMPH: Double

public final class TaborGolfRangeEngine: ObservableObject {
    @Published public var selectedClub: GolfClubType = .driver
    @Published public var lastShot: GolfShotTelemetry?
    @Published public var shotHistory: [GolfShotTelemetry] = []
    @Published public var targetPinYards: Double = 250.0
    
    public init() {}
    
    /// Converts a UnifiedThrowVector from the Tabor Tag (or phone swing) into real golf ball flight
    public func processGolfSwing(from vector: UnifiedThrowVector) -> GolfShotTelemetry {
        let specs = selectedClub.specs
        
        // 1. Clubhead Speed (MPH) scaled by shaft lever arm and swing intensity
        let rawSpeedScale = max(0.15, min(1.25, vector.launchSpeedNormalized))
        let maxClubMPH: Double = (selectedClub == .putter) ? 18.0 : 118.0
        let clubSpeedMPH = rawSpeedScale * maxClubMPH * (specs.shaftMeters / 1.14)
        
        // 2. Ball Speed (MPH)
        let ballSpeedMPH = clubSpeedMPH * specs.smashFactor
        
        // 3. Dynamic Launch Angle & Spin Rates
        let dynamicLoft = max(2.0, specs.baseLoftDeg + (vector.loftAngleDegrees * 0.18))
        let sideSpin = vector.lateralHookTorque * (selectedClub == .putter ? 50.0 : 1450.0)
        let backSpin = (selectedClub == .putter)
            ? 120.0
            : max(1800.0, (specs.baseLoftDeg * 165.0) + (clubSpeedMPH * 18.0))
        
        // 4. Carry & Total Distance Calculation (Yards)
        let carryYards: Double
        let totalYards: Double
        
        if selectedClub == .putter {
            carryYards = 0.0
            totalYards = max(1.0, (ballSpeedMPH * 1.45)) // Converts to green roll distance in yards (multiply x3 for feet)
        } else {
            let loftEfficiency = sin(2.0 * dynamicLoft * (.pi / 180.0))
            let aerodynamicCarry = (pow(ballSpeedMPH, 1.72) * max(0.35, loftEfficiency)) * 0.088
            carryYards = min(345.0, max(15.0, aerodynamicCarry))
            
            // Lower loft clubs get more fairway rollout; wedges bite and stop
            let rollRatio = max(0.02, (48.0 - specs.baseLoftDeg) / 160.0)
            totalYards = carryYards * (1.0 + rollRatio)
        }
        
        // 5. Lateral Curve & Shot Shape Classification
        let offlineYards = (sideSpin / 1450.0) * (totalYards * 0.14)
        let shapeLabel: String
        if abs(offlineYards) < 4.0 {
            shapeLabel = "Pure Straight"
        } else if offlineYards < -15.0 {
            shapeLabel = "Hard Hook"
        } else if offlineYards < 0 {
            shapeLabel = "Tight Draw"
        } else if offlineYards > 15.0 {
            shapeLabel = "High Slice"
        } else {
            shapeLabel = "Controlled Fade"
        }
        
        let telemetry = GolfShotTelemetry(
            club: selectedClub,
            clubheadSpeedMPH: clubSpeedMPH,
            ballSpeedMPH: ballSpeedMPH,
            launchAngleDeg: dynamicLoft,
            sideSpinRPM: sideSpin,
            backSpinRPM: backSpin,
            carryYards: carryYards,
            totalYards: totalYards,
            offlineYards: offlineYards,
            shotShapeLabel: shapeLabel
        )
        
        DispatchQueue.main.async {
            self.lastShot = telemetry
            self.shotHistory.insert(telemetry, at: 0)
        }
        
        return telemetry
    }
}

1. The Multi-Node Kinematic Chain (Equipment Tag + Body Tags)

Pairing a Tabor Tag on your equipment (golf club, bat, bowling ball, baseball, or basketball) with a set of Tabor Tags clipped to key body joints upgrades the system from a launch monitor into a full-body 3D biomechanical motion capture studio—without cameras, suits, or expensive lab gear.
By synchronizing all tags over Bluetooth Low Energy (BLE) on a unified 100Hz clock, the app reconstructs the kinematic sequence (how energy transfers from the ground up through the hips, torso, shoulder, wrist, and finally into the club or ball).

Recommended 5-Tag Athlete Body Mapping Set

• Tag 1 — Pelvis / Belt Buckle (NODE_PELVIS): Tracks hip rotation speed, sway, and early extension.
• Tag 2 — Lead Shoulder / Chest (NODE_TORSO): Measures shoulder turn, spine tilt, and X-Factor separation (hip-to-shoulder torque).
• Tag 3 — Lead Wrist / Forearm (NODE_WRIST): Captures wrist hinge, release timing, and forearm pronation/supination.
• Tag 4 — Lead Knee / Ankle (NODE_LOWER_LEAD): Monitors front-side bracing, weight transfer, and knee stability at impact or release.
• Tag 5 — Equipment Tag (NODE_EQUIPMENT): Snapped onto the golf club, bat, bowling ball, baseball, or basketball to measure final energy delivery.

2. How the Biomechanical Efficiency Engine Works

Instead of dumping raw numbers, the engine compares the live multi-tag telemetry against optimal athletic efficiency benchmarks for each sport:
• Golf Swing (golf_kinematics):
	• Kinematic Sequence Order: Checks whether peak rotational speed fires in the ideal Pelvis → Torso → Lead Arm → Club order.
	• Actionable Coaching Output: Detects casting (wrists firing before hips clear), reverse spine angle, or hip sway, then gives a single plain-English fix (e.g., "Start your downswing with your belt buckle—your wrists unhinged 45ms early, costing ~14 mph of clubhead speed").
• Baseball Pitching & Batting (baseball_kinematics):
	• Hip-Shoulder Separation: Measures maximum rotational stretch between NODE_PELVIS and NODE_TORSO before the arm whips forward.
	• Actionable Coaching Output: Flags arm-dragging or front-knee collapse (e.g., "Brace your lead knee at foot-strike—your front leg absorbed forward momentum instead of transferring it up into your shoulder").
• Bowling Release (bowling_kinematics):
	• Spine Tilt & Swing Plane: Tracks shoulder drop and wrist cup-to-release timing alongside the ball tag's rev rate.
	• Actionable Coaching Output: Spots muscling the ball or early thumb-exit (e.g., "Keep your shoulder steady through the flat spot—dropping your torso 8° early pulled your release 3 boards left").
• Basketball Shot Mechanics (basketball_kinematics):
	• Release Arc & Follow-Through Alignment: Compares wrist snap angle and elbow extension against the basketball tag's backspin RPM and launch angle.
	• Actionable Coaching Output: Identifies guide-hand interference or flat release (e.g., "Elevate your wrist release 6° higher—increasing launch arc from 41° to 48° widens your rim target area by 22%").

// MARK: - Body & Equipment Node Locations

public enum KinematicNodeRole: String, CaseIterable, Identifiable {
    case pelvis      = "Pelvis / Belt"
    case torso       = "Chest / Lead Shoulder"
    case leadWrist   = "Lead Wrist / Forearm"
    case leadKnee    = "Lead Knee / Leg"
    case equipment   = "Club / Bat / Ball"
    
    public var id: String { rawValue }
    
    /// Ideal firing order index in an efficient athletic kinetic chain (1 = first, 4 = last)
    public var idealSequenceOrder: Int {
        switch self {
        case .leadKnee:  return 1
        case .pelvis:    return 2
        case .torso:     return 3
        case .leadWrist: return 4
        case .equipment: return 5
        }
    }
}

public struct NodePeakTelemetry {

    public let peakAngularVelocityRadSec: Double
    public let timestampOfPeakSec: TimeInterval
    public let tiltAngleDegrees: Double
}

    public let efficiencyScore: Int              // 0 to 100% mechanical efficiency
    public let firingOrderActual: [KinematicNodeRole]

// MARK: - Multi-Tag Synchronized Biomechanics Engine

public final class TaborKinematicBodyEngine: ObservableObject {
    
    // Maps connected BLE peripheral UUIDs to assigned body joints or equipment
    @Published public var assignedNodes: [UUID: KinematicNodeRole] = [:]
    @Published public var latestReport: KinematicEfficiencyReport?
    @Published public var liveEfficiencyScore: Int = 0
    
    // Rolling peak capture per motion rep (windowed across 1.5s athletic motion)
    private var activeRepPeaks: [KinematicNodeRole: NodePeakTelemetry] = [:]
    private var repStartTime: TimeInterval = 0
    private var isCapturingRep: Bool = false
    
    public init() {}
    
    // MARK: - Tag Assignment
    
    public func assignTag(_ peripheralID: UUID, to role: KinematicNodeRole) {
        assignedNodes[peripheralID] = role
    }
    
    // MARK: - Multi-Node Telemetry Ingestion (Called at 100Hz per connected Tabor Tag)
    
    public func ingestNodeSample(peripheralID: UUID,
                                 accelG: SIMD3<Double>,
                                 gyroRadSec: SIMD3<Double>,
                                 pitchDeg: Double,
                                 timestamp: TimeInterval) {
        guard let role = assignedNodes[peripheralID] else { return }
        
        let angularSpeed = simd_length(gyroRadSec)
        let linearAccel = simd_length(accelG)
        
        // Trigger rep capture window when any node exceeds athletic initiation threshold
        if !isCapturingRep && (angularSpeed > 3.2 || linearAccel > 1.8) {
            isCapturingRep = true
            repStartTime = timestamp
            activeRepPeaks.removeAll()
        }
        
        guard isCapturingRep else { return }
        
        // Track peak rotational velocity and timestamp for each body segment
        if let existing = activeRepPeaks[role] {
            if angularSpeed > existing.peakAngularVelocityRadSec {
                activeRepPeaks[role] = NodePeakTelemetry(
                    role: role,
                    peakAngularVelocityRadSec: angularSpeed,
                    timestampOfPeakSec: timestamp,
                    tiltAngleDegrees: pitchDeg
                )
            }
        } else {
            activeRepPeaks[role] = NodePeakTelemetry(
                role: role,
                peakAngularVelocityRadSec: angularSpeed,
                timestampOfPeakSec: timestamp,
                tiltAngleDegrees: pitchDeg
            )
        }
        
        // Finalize & grade the full-body kinetic chain 450ms after initiation
        if (timestamp - repStartTime) >= 0.45 {
            isCapturingRep = false
            let report = evaluateKinematicChain(for: "Full-Body Athletic Motion")
            DispatchQueue.main.async {
                self.latestReport = report
                self.liveEfficiencyScore = report.efficiencyScore
            }
        }
    }
    
    // MARK: - Kinematic Sequence & Energy Transfer Solver
    
    public func evaluateKinematicChain(for sport: String) -> KinematicEfficiencyReport {
        // Sort captured nodes chronologically by the exact millisecond each reached peak speed
        let sortedByTime = activeRepPeaks.values.sorted { $0.timestampOfPeakSec < $1.timestampOfPeakSec }
        let actualOrder = sortedByTime.map { $0.role }
        
        // Check if body segments fired from ground up (Knee -> Pelvis -> Torso -> Wrist -> Equipment)
        var outOfOrderCount = 0
        for i in 1..<sortedByTime.count {
            if sortedByTime[i].role.idealSequenceOrder < sortedByTime[i - 1].role.idealSequenceOrder {
                outOfOrderCount += 1
            }
        }
        
        // Calculate X-Factor (Hip-to-Shoulder rotational separation)
        let pelvisPeak = activeRepPeaks[.pelvis]?.peakAngularVelocityRadSec ?? 6.0
        let torsoPeak = activeRepPeaks[.torso]?.peakAngularVelocityRadSec ?? 8.5
        let wristPeak = activeRepPeaks[.leadWrist]?.peakAngularVelocityRadSec ?? 14.0
        
        let velocityGainRatio = wristPeak / max(pelvisPeak, 1.0)
        let xFactorDeg = min(65.0, max(12.0, abs(torsoPeak - pelvisPeak) * 6.5))
        
        // Score mechanical efficiency (100 = textbook ground-up energy multiplication)
        var score = 96 - (outOfOrderCount * 14)
        if velocityGainRatio < 1.8 { score -= 12 } // Poor whip/lag transfer
        score = max(35, min(100, score))
        
        // Diagnose exact energy leak and generate actionable fix
        let bottleneck: String
        let advice: String
        
        if let wristTime = activeRepPeaks[.leadWrist]?.timestampOfPeakSec,
           let pelvisTime = activeRepPeaks[.pelvis]?.timestampOfPeakSec,
           wristTime < pelvisTime {
            let earlyMs = Int((pelvisTime - wristTime) * 1000.0)
            bottleneck = "Early Wrist Release (\(earlyMs)ms ahead of hips)"
            advice = "Initiate the move with your hips and belt buckle first—your hands fired \(earlyMs)ms early, leaking rotational power."
        } else if velocityGainRatio < 1.8 {
            bottleneck = "Low Whip Multiplication (Ratio: \(String(format: "%.1fx", velocityGainRatio)))"
            advice = "Brace your lead leg firmly at release so lower-body momentum snaps cleanly up into your torso and wrist."
        } else if let torsoTilt = activeRepPeaks[.torso]?.tiltAngleDegrees, abs(torsoTilt) > 38.0 {
            bottleneck = "Excessive Spine Tilt (\(Int(abs(torsoTilt)))°)"
            advice = "Keep your chest posture tall through impact—excessive spine tilt pulled your swing/release off plane."
        } else {
            bottleneck = "Optimal Ground-Up Energy Transfer"
            advice = "Textbook sequence: hips, torso, and wrist fired in order with \(String(format: "%.1fx", velocityGainRatio)) speed multiplication."
        }
        
        return KinematicEfficiencyReport(
            sportName: sport,
            efficiencyScore: score,
            firingOrderActual: actualOrder,
            isSequenceOptimal: outOfOrderCount == 0,
            xFactorSeparationDeg: xFactorDeg,
            primaryBottleneck: bottleneck,
            coachingAdjustment: advice
        )
    }
}
