import altair as alt
import pandas as pd

# Define the dimensions of the historical grid
eras = ['BC Era', 'AD Era', 'AC Era', 'Hey You Era']
categories = ['1. Legal/Bureaucracy', '2. Economic/Currency', '3. Social/Identity', '4. Spiritual/Mind']

# Populate the historical landmarks across structural infrastructure layers
landmarks = [
    # --- BC Era ---
    {'era': 'BC Era', 'category': '1. Legal/Bureaucracy', 'label': 'Imperial Decrees', 'desc': 'Seleucid decrees outlawing native tribal codes'},
    {'era': 'BC Era', 'category': '2. Economic/Currency', 'label': 'Tribute Tokens', 'desc': 'Centralized divine extraction currencies'},
    {'era': 'BC Era', 'category': '3. Social/Identity', 'label': 'Animal Branding', 'desc': 'Physical iron scarring marking state property'},
    {'era': 'BC Era', 'category': '4. Spiritual/Mind', 'label': 'Statue Bowing', 'desc': 'Mandatory allegiance rituals under death penalty'},
    # --- AD Era ---
    {'era': 'AD Era', 'category': '1. Legal/Bureaucracy', 'label': 'Quirinius Census', 'desc': 'Forced physical return for asset indexing'},
    {'era': 'AD Era', 'category': '2. Economic/Currency', 'label': 'Caesar Denarius', 'desc': 'Divine token required for market transaction'},
    {'era': 'AD Era', 'category': '3. Social/Identity', 'label': 'Charagma Stamp', 'desc': 'Official document seal and slave identification'},
    {'era': 'AD Era', 'category': '4. Spiritual/Mind', 'label': 'Incense Altars', 'desc': 'Civic loyalty oaths validating state dominion'},
    # --- AC Era ---
    {'era': 'AC Era', 'category': '1. Legal/Bureaucracy', 'label': 'Internal Passports', 'desc': 'Closed administrative spatial travel tracking'},
    {'era': 'AC Era', 'category': '2. Economic/Currency', 'label': 'Fiat Systems', 'desc': 'Sanitized global tender carrying systemic debt'},
    {'era': 'AC Era', 'category': '3. Social/Identity', 'label': 'Credit Scores', 'desc': 'Numerical algorithmic civic loyalty metrics'},
    {'era': 'AC Era', 'category': '4. Spiritual/Mind', 'label': 'Sub-Atomic UI Design', 'desc': 'Screen level optics driving behavior loops'},
    # --- Hey You Era ---
    {'era': 'Hey You Era', 'category': '1. Legal/Bureaucracy', 'label': 'Deplatforming Rails', 'desc': 'Automated spatial exile from digital networks'},
    {'era': 'Hey You Era', 'category': '2. Economic/Currency', 'label': 'Programmable CBDCs', 'desc': 'Central ledger assets with dynamic lockout hooks'},
    {'era': 'Hey You Era', 'category': '3. Social/Identity', 'label': 'Biometric Streams', 'desc': 'Continuous computer vision physical scans'},
    {'era': 'Hey You Era', 'category': '4. Spiritual/Mind', 'label': 'Predictive Feeds', 'desc': 'Dynamic custom optimization targeting cognition'}
]

df_landmarks = pd.DataFrame(landmarks)
df_landmarks['series'] = 'System Matrix Node'

# Generate the horizontal vector tracks tracking baseline continuity
vectors = []
for i, era in enumerate(eras):
    for cat in categories:
        vectors.append({
            'era': era,
            'category': cat,
            'order_id': i,
            'series': 'Convergence Vector'
        })
df_vectors = pd.DataFrame(vectors)

# Configure chart scales
x_scale = alt.Scale(domain=eras, padding=0.5)
y_scale = alt.Scale(domain=categories, padding=0.5)

# 1. Base Grid Matrix Boxes
grid_box = alt.Chart(df_landmarks).mark_rect(
    stroke='#2a2d3d',
    strokeWidth=1,
    fill='#111219',
    opacity=0.4
).encode(
    x=alt.X('era:N', scale=x_scale, title='Chronological Systemic Eras'),
    y=alt.Y('category:N', scale=y_scale, title='Infrastructure Layers')
).properties(
    width=900,
    height=550
)

# 2. Evolutionary Trajectory Vector Lines
vector_lines = alt.Chart(df_vectors).mark_line(
    color='#00ffcc',
    opacity=0.25,
    strokeWidth=1.5
).encode(
    x=alt.X('era:N', scale=x_scale),
    y=alt.Y('category:N', scale=y_scale),
    detail='category:N',
    order='order_id:Q'
)

# 3. System Milestone Nodes
milestone_nodes = alt.Chart(df_landmarks).mark_circle(
    size=120,
    color='#00ffcc',
    opacity=0.85
).encode(
    x=alt.X('era:N', scale=x_scale),
    y=alt.Y('category:N', scale=y_scale)
)

# 4. Primary POI Labels
label_text = alt.Chart(df_landmarks).mark_text(
    align='center',
    baseline='middle',
    dy=-20,
    color='#ffffff',
    fontSize=12,
    fontWeight='bold'
).encode(
    x=alt.X('era:N', scale=x_scale),
    y=alt.Y('category:N', scale=y_scale),
    text='label:N'
)

# 5. Supporting Subtext Descriptions
subtext_desc = alt.Chart(df_landmarks).mark_text(
    align='center',
    baseline='middle',
    dy=18,
    color='#8fa2b2',
    fontSize=9,
    lineBreak='\n'
).encode(
    x=alt.X('era:N', scale=x_scale),
    y=alt.Y('category:N', scale=y_scale),
    text='desc:N'
)

# 6. Focal Warning Frame for the Hey You Convergence Era
focal_box = alt.Chart(pd.DataFrame([{'era': 'Hey You Era', 'category': '2. Economic/Currency'}])).mark_rect(
    stroke='#ff3366',
    strokeWidth=2,
    fill='transparent'
).encode(
    x=alt.X('era:N', scale=x_scale),
    y=alt.Y('category:N', scale=y_scale)
)

# Layer charts sequentially to eliminate overlap
system_matrix_chart = alt.layer(
    grid_box,
    vector_lines,
    milestone_nodes,
    label_text,
    subtext_desc,
    focal_box
).configure_view(
    stroke='transparent',
    fill='#0a0b10'
).configure_axis(
    grid=False,
    domain=False,
    labelColor='#a0aab2',
    titleColor='#ffffff',
    labelFontSize=11,
    titleFontSize=13,
    titlePadding=15
)
